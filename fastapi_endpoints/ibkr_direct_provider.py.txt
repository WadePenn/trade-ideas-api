"""
fastapi_endpoints/services/ibkr_direct_provider.py
────────────────────────────────────────────────────
Live IBKR order provider — connects to TWS or IB Gateway via ib_insync.

Install:
    pip install ib_insync

TWS setup:
    1. File → Global Configuration → API → Settings
       ✓ Enable ActiveX and Socket Clients
       Socket port: 7497 (paper) or 7496 (live)
       ✓ Allow connections from localhost only

Gateway setup:
    Port: 4002 (paper) or 4001 (live)

Usage (in main.py startup):
    from fastapi_endpoints.services.ibkr_direct_provider import IBKRDirectProvider
    provider = IBKRDirectProvider(host="127.0.0.1", port=7497, client_id=1)
    if provider.connect():
        init_ibkr_order_service(provider)
    else:
        init_ibkr_order_service(InMemoryOrderProvider())   # safe fallback
"""
from __future__ import annotations

import logging
import threading
import time
from typing import Dict, Optional

log = logging.getLogger("fastapi_api.ibkr_direct")

try:
    from ib_insync import IB, ComboLeg, Contract, LimitOrder, util
    _HAS_IB = True
except ImportError:
    _HAS_IB = False
    log.warning(
        "ib_insync not installed — IBKRDirectProvider unavailable. "
        "Run: pip install ib_insync"
    )

from ..schemas.ibkr import (
    IBKRCancelResponse,
    IBKROrderStatusResponse,
    IBKRSubmitRequest,
    IBKRSubmitResponse,
    OrderStatus,
)


# ─────────────────────────────────────────────────────────────────
# Provider
# ─────────────────────────────────────────────────────────────────

class IBKRDirectProvider:
    """
    Routes orders directly to TWS / IB Gateway via ib_insync.

    Parameters
    ----------
    host      : TWS/Gateway host (default "127.0.0.1")
    port      : TWS paper=7497  live=7496 | Gateway paper=4002  live=4001
    client_id : Unique client ID — must differ from TWS's own client (default 1)
    timeout   : Connection timeout in seconds (default 10)
    readonly  : If True, connect in read-only mode (no orders placed)
    """

    def __init__(
        self,
        host:      str  = "127.0.0.1",
        port:      int  = 7497,
        client_id: int  = 1,
        timeout:   int  = 10,
        readonly:  bool = False,
    ) -> None:
        if not _HAS_IB:
            raise RuntimeError(
                "ib_insync not installed. Run: pip install ib_insync"
            )
        self._host      = host
        self._port      = port
        self._client_id = client_id
        self._timeout   = timeout
        self._readonly  = readonly
        self._ib        = IB()
        self._lock      = threading.Lock()
        self._connected = False

    # ── connection ─────────────────────────────────────────────────

    def connect(self) -> bool:
        """
        Attempt connection to TWS/Gateway.
        Returns True on success, False on failure (caller should fall back).
        """
        try:
            self._ib.connect(
                host      = self._host,
                port      = self._port,
                clientId  = self._client_id,
                timeout   = self._timeout,
                readonly  = self._readonly,
            )
            self._connected = True
            acc = self._ib.managedAccounts()
            log.info(
                "IBKRDirectProvider connected → %s:%d  accounts=%s",
                self._host, self._port, acc,
            )
            return True
        except Exception as exc:
            self._connected = False
            log.error(
                "IBKRDirectProvider failed to connect → %s:%d  error=%s",
                self._host, self._port, exc,
            )
            return False

    def disconnect(self) -> None:
        if self._connected:
            self._ib.disconnect()
            self._connected = False
            log.info("IBKRDirectProvider disconnected")

    @property
    def is_connected(self) -> bool:
        return self._connected and self._ib.isConnected()

    # ── submit ─────────────────────────────────────────────────────

    def submit(self, req: IBKRSubmitRequest) -> IBKRSubmitResponse:
        if not self.is_connected:
            raise ConnectionError("Not connected to TWS/Gateway")
        if self._readonly:
            raise PermissionError("Connected in read-only mode — orders disabled")
        if not req.legs:
            raise ValueError("Order must have at least one leg")

        with self._lock:
            # ── build the BAG combo contract ──────────────────────
            combo = Contract()
            combo.symbol   = req.symbol
            combo.secType  = "BAG"
            combo.currency = req.currency
            combo.exchange  = req.exchange

            combo_legs = []
            for leg in req.legs:
                # Resolve each option leg to a conId
                opt = Contract()
                opt.symbol    = leg.symbol
                opt.secType   = "OPT"
                opt.lastTradeDateOrContractMonth = leg.expiry.replace("-", "")[:8]
                opt.strike    = leg.strike
                opt.right     = leg.right        # "C" or "P"
                opt.exchange  = leg.exchange
                opt.currency  = req.currency

                details = self._ib.reqContractDetails(opt)
                if not details:
                    raise ValueError(
                        f"No contract found for {leg.symbol} "
                        f"{leg.right}{leg.strike} {leg.expiry}"
                    )
                con_id = details[0].contract.conId

                cl = ComboLeg()
                cl.conId     = con_id
                cl.ratio     = leg.ratio
                cl.action    = leg.action.value   # "BUY" or "SELL"
                cl.exchange  = leg.exchange
                combo_legs.append(cl)

            combo.comboLegs = combo_legs

            # ── build the limit order ─────────────────────────────
            order = LimitOrder(
                action    = req.action.value,       # "BUY" or "SELL"
                totalQuantity = req.quantity,
                lmtPrice  = req.limit_price,
            )
            order.account = req.account or ""
            order.tif     = req.time_in_force.value  # "DAY", "GTC", etc.
            order.orderRef = req.client_tag or ""

            # ── place ─────────────────────────────────────────────
            trade = self._ib.placeOrder(combo, order)
            self._ib.sleep(0.5)   # allow status update to arrive

            oid    = trade.order.orderId
            status = self._map_status(trade.orderStatus.status)

            log.info(
                "Order placed: id=%d sym=%s action=%s qty=%d lmt=%.2f status=%s",
                oid, req.symbol, req.action, req.quantity,
                req.limit_price, status,
            )
            return IBKRSubmitResponse(
                order_id = oid,
                account  = trade.order.account or req.account or "",
                status   = status,
            )

    # ── status ─────────────────────────────────────────────────────

    def get_status(self, order_id: int) -> IBKROrderStatusResponse:
        if not self.is_connected:
            raise ConnectionError("Not connected to TWS/Gateway")

        trades = self._ib.trades()
        for trade in trades:
            if trade.order.orderId == order_id:
                os = trade.orderStatus
                return IBKROrderStatusResponse(
                    order_id   = order_id,
                    status     = self._map_status(os.status),
                    filled_qty = int(os.filled),
                    remaining  = int(os.remaining),
                    avg_fill   = os.avgFillPrice,
                    message    = os.whyHeld or "",
                )

        return IBKROrderStatusResponse(
            order_id = order_id,
            status   = OrderStatus.INACTIVE,
            message  = "Order not found in active trades",
        )

    # ── cancel ─────────────────────────────────────────────────────

    def cancel(self, order_id: int) -> IBKRCancelResponse:
        if not self.is_connected:
            raise ConnectionError("Not connected to TWS/Gateway")

        trades = self._ib.trades()
        for trade in trades:
            if trade.order.orderId == order_id:
                self._ib.cancelOrder(trade.order)
                self._ib.sleep(0.5)
                log.info("Order cancelled: id=%d", order_id)
                return IBKRCancelResponse(
                    order_id = order_id,
                    status   = OrderStatus.CANCELLED,
                    message  = "Cancel request sent",
                )

        return IBKRCancelResponse(
            order_id = order_id,
            status   = OrderStatus.INACTIVE,
            message  = "Order not found — may already be filled or cancelled",
        )

    # ── helpers ────────────────────────────────────────────────────

    @staticmethod
    def _map_status(tws_status: str) -> OrderStatus:
        """Map TWS order status string → our OrderStatus enum."""
        mapping = {
            "Submitted":         OrderStatus.SUBMITTED,
            "PreSubmitted":      OrderStatus.SUBMITTED,
            "Filled":            OrderStatus.FILLED,
            "Cancelled":         OrderStatus.CANCELLED,
            "ApiCancelled":      OrderStatus.CANCELLED,
            "Inactive":          OrderStatus.INACTIVE,
            "PendingSubmit":     OrderStatus.PENDING,
            "PendingCancel":     OrderStatus.PENDING,
        }
        return mapping.get(tws_status, OrderStatus.PENDING)
